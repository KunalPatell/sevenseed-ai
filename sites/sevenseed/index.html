DEVELOPER SDK

Enterprise-Ready APIs & Python SDK

Integrate autonomous agent swarms, vector RAG retrieval, and secure BYOK execution into your enterprise stack in under 5 lines of code.

import os
from sevenseed import SevenseedClient
from sevenseed.agents import StateGraphSwarm

# 1. Initialize enterprise client with zero-data-retention BYOK
client = SevenseedClient(
    api_key=os.environ["SEVENSEED_API_KEY"],
    endpoint="https://api.sevenseed.in/v2"
)

# 2. Compile stateful multi-agent swarm
swarm = StateGraphSwarm(
    mode="venture_architect",
    model="groq/llama-3.3-70b-versatile",
    reflection_loops=2,
    enable_tools=["venture_intel", "financial_runway", "cybersecurity_recon"]
)

# 3. Execute mission with full state trace
result = swarm.invoke({
    "objective": "Evaluate Tier-2 fintech logistics opportunity with DPDP compliance"
})

print(f"Consensus Score: {result.critique_score}/100")
print(f"Deliverable: {result.deliverable.markdown}")
