# Kahn's Algorithm for Agent Dependency
def solve_topo(graph):
    in_deg = {u: 0 for u in graph}
    for u in graph:
        for v in graph[u]: in_deg[v] += 1
    queue = [u for u in graph if in_deg[u] == 0]
    return len(queue) > 0
